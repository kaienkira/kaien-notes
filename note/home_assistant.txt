===============================================================================
*. 首次启动配置WIFI
===============================================================================
    mount /dev/sdb1 <mount_dir>
    mkdir <mount_dir>/CONFIG/network
    vi <mount_dir>/CONFIG/network/my-network
        [connection]
        id=my-network
        uuid=<uuid(use uuidgen)>
        type=802-11-wireless

        [802-11-wireless]
        mode=infrastructure
        ssid=<wifi_ssid>

        [802-11-wireless-security]
        auth-alg=open
        key-mgmt=wpa-psk
        psk=<wifi_password>

        [ipv4]
        method=auto

        [ipv4]
        method=manual
        address=192.168.1.111/24;192.168.1.1
        dns=8.8.8.8;8.8.4.4;

    mount /dev/sdb7 <mount_dir>
    vi <mount_dir>/etc/NetworkManager/conf.d/wifi-powersave-off.conf
        [connection]
        wifi.powersave = 2
        # disable = 2 enable = 3
    chmod 600 wifi-powersave-off.conf
