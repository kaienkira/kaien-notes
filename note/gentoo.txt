*. 更新包数据库
    emerge --sync
*. 更新所有包
    emerge -uDN @world

    emerge -e @world
*. 清理软件包
    emerge -c(--depclean)
    eclean -d distfiles
*. 卸载包
    emerge -W(--deselect) <package_name> && emerge -c
*. 查找所有包
    emerge -s(--search) <package_name>
    emerge -S(--searchdesc) <package_name>
    qsearch <package_name>
*. 查看包信息
    equery meta(m) <package_name>
*. 查看包依赖
    equery depgraph(g) <package_name>
    qdepends -v <package_name>
*. 查看包反向依赖
    equery depends(d) <package_name>
    qdepends -Qv <package_name>
*. 查找已安装的包
    qlist -I <keyword>
*. 查看安装的所有包
    qlist -Iv
*. 查看world包含的包
    cat /var/lib/portage/world
*. 查看包安装的文件
    equery files <package_name>
    qlist <package_name>
*. 查找文件对应哪个包
    equery belongs(b) <file_path>
    qfile <file_path>
*. 查看包占用的大小
    equery size(s) <package_name>
    qsize -f <package_name>
*. 查看所有包占用的大小
    qsize -Sf
*. 查看包的当前USE
    equery use(u) <package_name>
*. 查看包的所有USE
    quse -Dvp <package_name>

*. 计算blake2b hash
    echo -n '<file_name>' | b2sum

===============================================================================
*. 依赖类型
     DEPEND -> Build dependencies 编译依赖 (CHOST)
    BDEPEND -> Build dependencies 编译依赖 (CBUILD)
    RDEPEND -> Runtime dependencies 运行时依赖
    PDEPEND -> Post dependencies 类似运行时依赖
               不要求在本包之前安装, 用于解决循环依赖
    IDEPEND -> Install dependencies 安装时依赖
