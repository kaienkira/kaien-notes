===============================================================================
-*- Network Settings -*-
===============================================================================
*. recv buffer
    /proc/sys/net/core/rmem_default
    /proc/sys/net/core/rmem_max

*. send buffer
    /proc/sys/net/core/wmem_default
    /proc/sys/net/core/wmem_max
    /proc/sys/net/ipv4/tcp_wmem (min default max)

*. time_wait status
    net.ipv4.tcp_tw_recycle
        Enabling this option is not recommended since this causes problems
        when working with NAT
    net.ipv4.tcp_tw_reuse
        server_side useless with for incoming connections
        client_side almost-safe solution an outgoing connection in the TIME-WAIT
                    state can be reused after just one second
    net.ipv4.tcp_fin_timeout
        FIN_WAIT_2 status timeout

*. max open sockets
    ulimit -n
    cat /proc/sys/fs/file-nr
    cat /proc/sys/fs/file-max
    cat /proc/sys/net/ipv4/ip_local_port_range 
    cat /proc/sys/net/ipv4/ip_local_reserved_ports

*. extend lv

p -> l -> t -> 8e

vgs (see size)
lvs (see size)

xfs_growfs /dev/<vg_name>/<root> (for xfs)

*. reduce lv
# reboot use rescue mode, skip mount
pvscan
vgscan
vgchange -a y
lvscan

resize2fs -p /dev/<vg_name>/<root> <new_size>
lvreduce -L <new_size> /dev/<vg_name>/<root>

===============================================================================
-*- udev -*-
-------------------------------------------------------------------------------
*. 修改硬盘名字
udevadm info -a -p $(udevadm info -q path -n </dev/sda>)
vi /etc/udev/rules.d/00-disk.rules
    KERNEL=="sda", NAME="vda"
    KERNEL=="sda?", NAME="vda%n"
    KERNEL=="sdb", NAME="vdb"
    KERNEL=="sdb?", NAME="vdb%n"
*. 等待内核处理完udev事件
udevadm settle
    settle waits for udevd to finish processing any uevents queued
    by the kernel when it is called.
