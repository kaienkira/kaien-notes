*. packages
    vi /usr/local/etc/pkg.conf
        PKG_ENV {
            all_proxy = "socks5://<ip>:<port>"
        }

    pkg bootstrap
    pkg info
    pkg update -f
    pkg install <package_name>

*. ssh
    vi /etc/ssh/sshd_config
        PermitRootLogin yes
        PasswordAuthentication yes
    service sshd restart
