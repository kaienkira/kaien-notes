*. run in vmware
    pacman -S qemu
    gzip -kd openwrt-*-x86-64-generic-ext4-combined.img.gz
    dd bs=1024M count=1 conv=sync if=openwrt-*-x86-64-generic-ext4-combined of=openwrt.img
    qemu-img convert -f raw -O vmdk openwrt.img openwrt.vmdk

*. opkg
    apk update
    apk info
    apk upgrade
    apk search <package>

*. opkg mirror
    vi /etc/opkg/distfeeds.conf

*. luci
    apk update
    apk add luci luci-ssl

*. capture packets
    apk add tcpdump
    killall tcpdump; tcpdump -n -i eth1
    ssh user@myledebox tcpdump -i eth1 -U -s0 -w - 'not port 22' | sudo XAUTHORITY=~/.Xauthority wireshark -k -i -

*. aircrack-ng
    apk add aircrack-ng
    apk add airmon-ng
    iw phy
    iw dev
    airodump-ng -i -c <channel> --bssid <bssid> -w <file> <dev>
    aireplay-ng -0 <send_count> -a <ap_mac_addr> -c <client_mac_addr> <dev> 
    aircrack-ng -w <password_list_file> -b <bssid> <cap_file>

* usb
    apk add kmod-usb-core
    apk add kmod-usb-storage
    # usb1.1
    apk add kmod-usb-uhci
    apk add kmod-usb-ohci
    # usb2.0
    apk add kmod-usb2
    # usb3.0
    apk add kmod-usb3
    # utils
    apk add usbutils
    # exfat support
    apk add kmod-fs-exfat
    # mount
    mount /dev/sda1 /mnt/usb
