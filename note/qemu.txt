qemu-img create -f qcow2 <img_file> 20G
qemu-img info <img_file>

===============================================================================
-*- machine -*-
===============================================================================
-machine type=q35,smm=on,accel=kvm
-m 4096 -smp 8 -cpu host

===============================================================================
-*- uefi -*-
===============================================================================
-bios OVMF.fd

-drive if=pflash,format=raw,unit=0,file=/usr/share/OVMF/OVMF_CODE_4M.ms.fd,readonly=on
-drive if=pflash,format=raw,unit=1,file=./OVMF_VARS_win11.fd

-chardev socket,id=chrtpm,path=./tpm/swtpm.sock
-tpmdev emulator,id=tpm0,chardev=chrtpm
-device tpm-crb,tpmdev=tpm0

////////////////////////////////////////////////////////////////////////////////
brew install swtpm
ln -s /home/linuxbrew/.linuxbrew/bin/gnutls-certtool ~/bin/certtool

swtpm_setup --tpm2 --tpmstate tpm \
    --create-ek-cert --create-platform-cert --ecc \
    --lock-nvram

cp /usr/share/OVMF/OVMF_VARS_4M.ms.fd ./OVMF_VARS_win11.fd

swtpm socket --tpm2 --tpmstate dir=tpm \
    --ctrl type=unixio,path=tpm/swtpm.sock \
    --daemon \
    --terminate

-*- spice -*-
===============================================================================
-device virtio-serial-pci
-object secret,id=sec0,format=raw,file=$HOME/local/vm/tls-key/login_pass
-spice port=5901,disable-ticketing=off,password-secret=sec0,x509-dir=$HOME/local/vm/tls-key/

-spice port=5901,disable-ticketing=on

-spice port=5901,disable-ticketing=off,password=abc123,x509-dir=<tls_dir>

-device virtserialport,chardev=spicechannel0,name=com.redhat.spice.0
-chardev spicevmc,id=spicechannel0,name=vdagent

////////////////////////////////////////////////////////////////////////////////
openssl genrsa -out ca-key.pem 2048
openssl req -new -x509 -days 3650 -key ca-key.pem -out ca-cert.pem \
    -subj "/CN=SPICE CA"

openssl genrsa -out server-key.pem 2048
openssl req -new -key server-key.pem -out server-csr.pem -subj "/CN=myhost"
openssl x509 -req -days 3650 -in server-csr.pem \
    -CA ca-cert.pem -CAkey ca-key.pem -CAcreateserial -out server-cert.pem

////////////////////////////////////////////////////////////////////////////////
pacman -S spice-vdagent
pacman -S wl-clipboard xclip
