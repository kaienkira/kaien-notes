# virglrenderer
git clone https://gitlab.freedesktop.org/virgl/virglrenderer.git
git checkout 1.3.0
meson setup build \
    --prefix=$HOME/local/qemu \
    --buildtype=release \
    -Dlibdir=lib \
    -Dplatforms=egl,glx \
    -Dvenus=true
cd build && ninja -j8 && ninja install

# aemu
git clone https://android.googlesource.com/platform/hardware/google/aemu
git checkout v0.1.2-aemu-release
cmake -DAEMU_COMMON_GEN_PKGCONFIG=ON \
      -DAEMU_COMMON_BUILD_CONFIG=gfxstream \
      -DENABLE_VKCEREAL_TESTS=OFF \
      --install-prefix "$HOME/local/qemu" \
      -B build
cmake --build build -j8
cmake --install build --prefix "$HOME/local/qemu"

# gfxstream
git clone https://android.googlesource.com/platform/hardware/google/gfxstream
git checkout v0.1.2-gfxstream-release
meson setup build \
    --prefix=$HOME/local/qemu \
    --buildtype=release
    -Dlibdir=lib \
    -Ddefault_library=static \
    -Ddecoders=gles,vulkan,composer
cd build && ninja -j8 && ninja install

# rutabaga_ffi
git clone https://github.com/magma-gpu/rutabaga_gfx.git
git checkout v0.1.75
meson setup build \
    --prefix=$HOME/local/qemu \
    --buildtype=release \
    -Dlibdir=lib \
    -Dffi=true \
    -Dfeatures=gfxstream
cd build && ninja -j8 && ninja install

# qemu
git clone https://gitlab.com/qemu-project/qemu.git
git checkout v11.0.1
mkdir build && cd build
../configure \
    --prefix=$HOME/local/qemu \
    --target-list=x86_64-softmmu \
    --enable-kvm \
    --enable-opengl \
    --enable-virglrenderer \
    --enable-rutabaga-gfx \
    --enable-spice \
    --enable-spice-protocol \
    --enable-slirp
make -j8 && make install
