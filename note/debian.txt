*. samba
    sudo apt-get install samba
    useradd -s /sbin/nologin sambaadmin
    useradd -s /sbin/nologin sambaguest
    smbpasswd -a sambaadmin
    smbpasswd -a sambaguest
    vi /etc/samba/smb.conf
        [global]
        workgroup = WORKGROUP
        interfaces = 192.168.10.10/24
        hosts allow = 192.168.10.10/24
        log file = /var/log/samba/samba.log
        max log size = 1000
        logging = file

        server role = standalone server
        obey pam restrictions = no
        unix password sync = no
        pam password change = no
        map to guest = bad user

        [share]
        comment = File Share
        path = /opt/samba
        browseable = yes
        read only = yes
        write list = sambaadmin
        create mask = 0755
        guest ok = no
        valid users = sambaadmin,sambaguest
    systemctl restart smbd

*. dns
    vi /etc/dhcp/dhclient.conf
        supersede domain-name-servers 223.5.5.5, 119.29.29.29, 8.8.8.8;

*. 查看包信息
    dpkg-query -s <package_name>
*. 查看包文件
    dpkg-query -L <package_name>
*. 列出所有安装的软件包
    dpkg -l
*. 查找文件属于哪个包
    dpkg -S <file>
*. 彻底删除软件包
    sudo apt-get --purge autoremove packagename
*. 清除所有被标记为rc的软件包配置
    apt purge '~c'
*. 列出编译依赖
    apt-get -s build-dep <package>
*. 列出包属于哪个源
    grep -l <package> /var/lib/apt/lists/*
*. 列出手动安装的包
    apt-mark showmanual
*. 把包设置为自动安装
    apt-mark auto <package>

*. network
    vi /etc/network/interface
        iface ens32 inet dhcp
        iface ens32 inet static
        address <ip>
        netmask <netmask>
        gateway <gateway>
        dns-nameservers <dns1> <dns2>

*. proxy
    systemctl enable systemd-networkd.service
    systemctl mask ifup@.service
    systemctl mask networking.service
    vi /etc/systemd/network/10-wired.network
        [Match]
        Name=en*

        [Network]
        DHCP=ipv4
    vi /etc/sysctl.d/ip_forward.conf
        net.ipv4.ip_forward=1
        net.ipv6.conf.all.forwarding=1

    apt-get install firmware-iwlwifi

*. qemu
    apt install qemu-utils qemu-system-x86
