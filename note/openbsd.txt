*. 设置镜像
    vi /etc/installurl
        https://mirrors.tuna.tsinghua.edu.cn/OpenBSD
        # https://cdn.openbsd.org/pub/OpenBSD
*. 更新系统
  syspatch
  sysupgrade
  sysmerge -d

*. 搜索包
  pkg_info -Q <keyword>
*. 安装包
  pkg_add <package_name>
*. 更新包
  pkg_add -Uu
*. 删除包
  pkg_delete -a <pacakge_name>

*. 关机
  halt -p

*. 修改磁盘加密密码
  bioctl -P <sd0>
===============================================================================
*. ksh
    vi ~/.profile
        export ENV=$HOME/.kshrc
===============================================================================
*. show listening ports
    netstat -nl -f inet
    netstat -nl -f inet6
*. sshd
    rcctl enable sshd
    rcctl start sshd
===============================================================================
*. static network address
    vi /etc/hostname.em0
        inet 192.168.100.11/24 # inet autoconf
    vi /etc/mygate
        192.168.100.2
    vi /etc/resolv.conf
        nameserver 8.8.8.8
    sh /etc/netstart em0
===============================================================================
*. wifi
  fw_update
  vi /etc/hostname.<wifi_interface>
    nwid <wifi_ssid> wpakey <wifi_key> 
    autoconf
  sh /etc/netstart <wifi_interface>
